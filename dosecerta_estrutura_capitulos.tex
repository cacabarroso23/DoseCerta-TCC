\cs_new_protected:Npn \__dosecerta_organizacao:
  {
    % Trocar somente o parágrafo de organização, preservando o restante da introdução.
    \regex_replace_once:nnN
      { \c{section} \{ Organização \s+ do \s+ [Tt]rabalho \} }
      { \c{DoseCertaInicioOrganizacao} } \l__dosecerta_fonte_tl
    \tl_if_in:NnTF \l__dosecerta_fonte_tl {\DoseCertaInicioOrganizacao}
      {
        \seq_set_split:NnV \l__dosecerta_org_seq
          {\DoseCertaInicioOrganizacao} \l__dosecerta_fonte_tl
        \seq_pop_left:NN \l__dosecerta_org_seq \l__dosecerta_prefixo_tl
        \seq_pop_left:NN \l__dosecerta_org_seq \l__dosecerta_sufixo_tl
        \seq_set_split:NnV \l__dosecerta_resto_seq
          {\section} \l__dosecerta_sufixo_tl
        \seq_pop_left:NN \l__dosecerta_resto_seq \l__dosecerta_descartado_tl
        \tl_set_eq:NN \l__dosecerta_fonte_tl \l__dosecerta_prefixo_tl
        \tl_put_right:Nn \l__dosecerta_fonte_tl
          {
            \section{Organização~do~trabalho}
            O~trabalho~está~organizado~em~cinco~capítulos.
            O~Capítulo~\ref{cap:introducao}~apresenta~o~problema,~os~objetivos,
            a~justificativa~e~a~delimitação.
            O~Capítulo~\ref{cap:base_teorica}~reúne~a~fundamentação~teórica~e~o~estado~da~arte.
            O~Capítulo~\ref{cap:metodo_integrado}~descreve~a~metodologia,~os~requisitos,
            a~arquitetura,~as~frentes~técnicas~e~os~procedimentos~de~integração~e~testes.
            O~Capítulo~\ref{cap:resultados_discussao}~reúne~os~resultados~e~sua~discussão.
            O~Capítulo~\ref{cap:conclusao}~apresenta~as~considerações~finais~e~as~possibilidades~de~continuidade.
            As~referências~e~os~elementos~pós-textuais~complementam~a~monografia.
          }
        \seq_map_inline:Nn \l__dosecerta_resto_seq
          { \tl_put_right:Nn \l__dosecerta_fonte_tl {\section ##1} }
      }
      {
        \PackageWarning{dosecerta-estrutura}
          {Titulo~Organizacao~do~trabalho~nao~localizado;~conferir~introducao}
      }
  }

\cs_new_protected:Npn \__dosecerta_rebaixar:
  {
    % Se o arquivo já começa como seção, não rebaixá-lo novamente.
    \tl_if_in:NnT \l__dosecerta_fonte_tl {\chapter}
      {
        \tl_replace_all:Nnn \l__dosecerta_fonte_tl {\subparagraph} {\DoseCertaTituloSubparagrafo}
        \tl_replace_all:Nnn \l__dosecerta_fonte_tl {\paragraph} {\DoseCertaTituloSubparagrafo}
        \tl_replace_all:Nnn \l__dosecerta_fonte_tl {\subsubsection} {\DoseCertaTituloParagrafo}
        \tl_replace_all:Nnn \l__dosecerta_fonte_tl {\subsection} {\DoseCertaTituloSubsubsecao}
        \tl_replace_all:Nnn \l__dosecerta_fonte_tl {\section} {\DoseCertaTituloSubsecao}
        \tl_replace_all:Nnn \l__dosecerta_fonte_tl {\chapter} {\DoseCertaTituloSecao}
        \tl_replace_all:Nnn \l__dosecerta_fonte_tl {Este~capítulo} {Esta~seção}
        \tl_replace_all:Nnn \l__dosecerta_fonte_tl {este~capítulo} {esta~seção}
        \tl_replace_all:Nnn \l__dosecerta_fonte_tl {O~capítulo~seguinte} {A~seção~seguinte}

\cs_new_protected:Npn \__dosecerta_ler:nn #1#2
  {
    \group_begin:
    \tl_set:Nn \l__dosecerta_arquivo_tl {#2}
    \regex_replace_once:nnN { \A \./ } {} \l__dosecerta_arquivo_tl
    \regex_replace_once:nnN { \.tex \Z } {} \l__dosecerta_arquivo_tl
    \tl_put_right:Nn \l__dosecerta_arquivo_tl {.tex}
    \prop_if_in:NVTF \g__dosecerta_lidos_prop \l__dosecerta_arquivo_tl
      { \PackageInfo{dosecerta-estrutura}{Inclusao~repetida~ignorada:~#2} }
      {
        \file_if_exist:VTF \l__dosecerta_arquivo_tl
          {
            \prop_gput:NVn \g__dosecerta_lidos_prop \l__dosecerta_arquivo_tl {lido}
            % Ler com catcodes normais para preservar espaços, comentários e comandos.
            \file_get:VnN \l__dosecerta_arquivo_tl {\ExplSyntaxOff} \l__dosecerta_fonte_tl
            \__dosecerta_referencias:
            \str_if_eq:nnT {#2} {Capitulos/01_introducao}
              { \__dosecerta_organizacao: }
            \str_if_eq:nnT {#1} {secao}
              { \__dosecerta_rebaixar: }
            % Inclusões internas passam pela mesma guarda (tabelas, 10b e desenhos).
            \str_if_eq:nnTF {#1} {secao}
              {
                \tl_replace_all:Nnn \l__dosecerta_fonte_tl {\input} {\DoseCertaSecaoArquivo}
                \tl_replace_all:Nnn \l__dosecerta_fonte_tl {\include} {\DoseCertaSecaoArquivo}
              }
              {
                \tl_replace_all:Nnn \l__dosecerta_fonte_tl {\input} {\DoseCertaArquivoIntegral}
                \tl_replace_all:Nnn \l__dosecerta_fonte_tl {\include} {\DoseCertaArquivoIntegral}
              }
            \tl_use:N \l__dosecerta_fonte_tl
          }
          {
            \PackageError{dosecerta-estrutura}{Arquivo~necessario~ausente:~#2}
              {Este~ZIP~e~de~substituicao.~Mantenha~os~fontes~tecnicos~do~projeto~original.}
          }
      }
    \group_end:
  }
